# =========================================================
#  SLOTH feature showcase
#  A lazy, Haskell-like, untyped (JS-like) scripting language.
#  Run:  deno run --allow-read main.ts features.txt
#  NOTE: indentation is strict "4-space layout" — every nested
#  level is exactly +4 spaces (and only +4).
# =========================================================

# ---------------------------------------------------------
# 1. DATA TYPES (variants / ADTs)
#    Unary constructor `Just` wraps a value; bare `Nothing` is a tag.
# ---------------------------------------------------------
data Maybe a = Just a | Nothing

debug (Just 42)        # => Just 42
debug Nothing          # => Nothing

# ---------------------------------------------------------
# 2. FUNCTIONS & LAMBDAS
#    Two styles to define a function:
#      - lambda:     name = x -> body   (multi-param: a b c -> body)
#      - Haskell:    name x = body      (params attached to the left of `=`)
#    Both support recursion and partial application.
# ---------------------------------------------------------
func x = x + x                # Haskell style
debug (func 21)               # => 42

double = x -> x * 2           # lambda style
debug (double 21)             # => 42

add a b = a + b               # Haskell style, multi-param
debug (add 3 4)               # => 7

addTwo = add 2                # partial application
debug (addTwo 10)             # => 12

# ---------------------------------------------------------
# 3. RECURSION
# ---------------------------------------------------------
factorial n = match n [
    0 -> 1
    _ -> n * factorial (n - 1)
]
debug (factorial 10)   # => 3628800

fib n = match n [
    0 -> 0
    1 -> 1
    _ -> fib (n - 1) + fib (n - 2)
]
debug (fib 10)         # => 55

# ---------------------------------------------------------
# 4. PATTERN MATCHING (match) & PATTERN DEFINITIONS
#    Patterns: literals, wildcard _, cons x:rest, lists, variants, guards.
#    `match` is THE pattern-matching form (there is no `case`):
#        match scrutinee [ pattern -> expr, ... ]
#    A `[x, y]` pattern matches BOTH a list (positionally) and a record
#    (by field name). A def can also be given as multiple clauses (equations)
#    like Haskell, dispatched in order on its parameters:
#         f pat1 = body1
#         f pat2 = body2
# ---------------------------------------------------------
data Maybe a = Just a | Nothing

safeHead xs = match xs [
    [] -> Nothing
    x:rest -> Just x
]
debug (safeHead [1, 2, 3])   # => Just 1
debug (safeHead [])          # => Nothing

# multiple equations (clauses), like Haskell
len [] = 0
len (x:rest) = 1 + len rest
debug (len [1, 2, 3])        # => 3

sumList xs = match xs [
    [] -> 0
    x:rest -> x + sumList rest
]
debug (sumList [1, 2, 3, 4, 5])  # => 15

# exact-length list pattern — binds each element
pairSum xs = match xs [
    [a, b] -> a + b
    _ -> 0
]
debug (pairSum [10, 20])   # => 30
debug (pairSum [1])        # => 0

# the same `[x, y]` pattern also destructures a RECORD by field name
sumRec [x, y] = x + y
debug (sumRec [x: 10, y: 20])  # => 30
debug (sumRec [10, 20])        # => 30

# guards
classify n = match n [
    x | x < 0 -> "negative"
    x | x == 0 -> "zero"
    _ -> "positive"
]
debug (classify (-5))   # => "negative"
debug (classify 0)      # => "zero"
debug (classify 7)      # => "positive"

# ---------------------------------------------------------
# 4b. MATCH LAYOUTS —  match scrutinee [ pattern -> expr, ... ]
#     Alternatives are delimited with [ ], separated by `,` OR a newline.
#     PREFERRED STYLE for a function whose entire body is one match is the
#     same-line-open form:
#         f x = match x [
#             pattern -> body
#         ]
#     (The block form `f x =\n    match x [ ... ]` also parses but is not
#     used in this showcase.) The fully single-line form:
#         f x = match x [ pat1 -> e1, pat2 -> e2 ]
# ---------------------------------------------------------
factA n = match n [
    0 -> 1
    _ -> n * factA (n - 1)
]
debug (factA 10)            # => 3628800

factC n = match n [0 -> 1, _ -> n * factC (n - 1)]
debug (factC 6)             # => 720

# all patterns work here too: cons, literals, _ , guards, ...
safeM xs = match xs [
    [] -> Nothing
    x:rest -> Just x
]
debug (safeM [3, 4])        # => Just 3
debug (safeM [])            # => Nothing

classM n = match n [
    x | x < 0 -> "negative"
    x | x == 0 -> "zero"
    _ -> "positive"
]
debug (classM (-5))         # => "negative"
debug (classM 0)            # => "zero"
debug (classM 7)            # => "positive"

# a def body may also start `match` on the SAME line as `=`
matchInline n = match n [
    0 -> "zero"
    _ -> "other"
]
debug (matchInline 0)       # => "zero"
debug (matchInline 9)       # => "other"

# ---------------------------------------------------------
# 5. LISTS — heterogeneous, like JS arrays, but almost lazy.
#    Literal lists stay in an unevaluated (lazy) state until used. A list of
#    bare names is still a list — the names resolve to the variables' values.
#    (An OBJECT is a record literal: [x: 1, y: 2, z: 3].)
# ---------------------------------------------------------
debug [213, "hello"]          # => [213, "hello"]
debug [1, 2, 3]               # => [1, 2, 3]
debug ([] ++ [1, 2] ++ [3])   # => [1, 2, 3]

# multi-line list literals: elements separate on their own lines, so commas
# are optional there (a comma also still works):
arr = [
    1
    2
    3
]
debug arr                     # => [1, 2, 3]
arr = [
    1,  2,  3
]
debug arr                     # => [1, 2, 3]   (commas on one line fine too)

nested = [
    1
    [2, 3]
    4
]
debug nested                  # => [1, [2, 3], 4]

[1, 2, 3] $ head $ debug   # => 1
debug (tail [1, 2, 3])        # => [2, 3]
debug (length "hello")        # => 5
debug (length [1, 2, 3])      # => 3
debug (null [])               # => true
debug (null [1])              # => false

# a bare-name list is ALWAYS a list — elements are (lazy) variable refs
a = 100
b = 200
vals = [a, b]
debug vals                       # => [100, 200]
debug (map (x -> x * 2) vals)    # => [200, 400]

# ---------------------------------------------------------
# 6. RECORDS — [name: value, ...] and field access
# ---------------------------------------------------------
person = [name: "Alice", age: 30]
debug person.name        # => "Alice"
debug person.age         # => 30
debug person             # => [name: "Alice", age: 30]

# nested records + chained access
nested = [inner: [z: 99]]
debug nested.inner.z     # => 99

# ---------------------------------------------------------
# 7. PROTOTYPE-BASED OBJECTS —  proto.[overrides] CLONES the object
#    The prototype is any RECORD (LOWERCASE name — capital names are
#    constructors); .[values] replaces values positionally, .[f: v, ...]
#    overrides by name (and may ADD new fields). A bare-name list is a LIST,
#    never an object.
# ---------------------------------------------------------
vector3 = [x: 0, y: 0, z: 0]        # prototype with defaults
debug vector3                       # => [x: 0, y: 0, z: 0]
debug vector3.[1, 2, 3]             # => [x: 1, y: 2, z: 3]   positional clone
debug vector3.[x: 1, y: 2, z: 3]    # => [x: 1, y: 2, z: 3]   named override
debug vector3.[x: 1, y: 2]          # => [x: 1, y: 2, z: 0]   partial — z kept
debug vector3.[x: 1, y: 2, z: 3, a: 0]  # => [x: 1, y: 2, z: 3, a: 0]  new field
debug vector3                       # => [x: 0, y: 0, z: 0]   prototype untouched
debug vector3.[1, 2, 3].z           # => 3     (.z chains over the clone)

# ---------------------------------------------------------
# 8. TERNARY — cond ? a : b  (replaces if/then/else)
# ---------------------------------------------------------
max a b = a > b ? a : b
debug (max 4 9)            # => 9

smallest a b c = (a < b ? a : b) < c ? (a < b ? a : b) : c
debug (smallest 7 2 9)     # => 2

# ---------------------------------------------------------
# 9. BINDINGS —  symbol = value  (the ONLY form; no let, no mutation)
#    The LHS may be a destructuring PATTERN:  [name, hello] = expr
#    (list positionally, record by field name). Bindings are scoped
#    to their block; a block's value is its LAST statement.
# ---------------------------------------------------------
greeting = {
    name = "world"
    hello = "Hello"
    hello ++ ", " ++ name ++ "!"
}
debug greeting             # => "Hello, world!"

greeting = {
    [name, hello] = ["world", "Hello"]
    hello ++ ", " ++ name ++ "!"
}
debug greeting             # => "Hello, world!"

# Constructor patterns destructure like Haskell — bare (Just v = ...),
# parenthesized ((Just v) = ...), or nested in a list ([Just v] = [...]).
# ONLY a DECLARED constructor (via data) can appear capital on the left of =;
# a capital name is NEVER a value/function name (Template = ... is an error —
# use lowercase, and build objects with the prototype clone from section 7):
data Maybe a = Just a | Nothing
Just v = Just 123
debug v                    # => 123

# ---------------------------------------------------------
# 10. PIPES, BACKTICK INFIX, OPERATOR SECTIONS
#    x $ f            feeds x into f   (\value-left pipe, like F# |>)
#    a `f` b         == f a b          (backtick infix, like Haskell)
#    (+ 2) / (2 +)    operator sections  ->  \x -> x + 2 / \x -> 2 + x
# ---------------------------------------------------------
negate x = -x
debug (5 $ negate)                        # => -5
debug ([1, 2, 3] $ map (x -> x * 10))     # => [10, 20, 30]
[1, 2, 3] $ map (x -> x * 10) $ toString $ print  # => [10, 20, 30]

myApp xs ys = xs ++ ys
debug ([1, 2] `myApp` [3, 4])             # => [1, 2, 3, 4]

debug (map (+ 2) [1, 2, 3])               # => [3, 4, 5]
debug (map (2 +) [1, 2, 3])               # => [3, 4, 5]
debug (map (* 10) [1, 2, 3])              # => [10, 20, 30]
debug (filter (2 <) [3, 1, 4, 1, 5])      # => [4, 5]
debug (-2)                                # => -2  ((- 2) is NOT a section:
                                          #     unary minus wins; use a lambda)

# ---------------------------------------------------------
# 11. HIGHER-ORDER FUNCTIONS — map / filter / foldl / foldr
# ---------------------------------------------------------
nums = [1, 2, 3, 4, 5]
debug (map (x -> x * x) nums)                 # => [1, 4, 9, 16, 25]
debug (filter (x -> x > 2) nums)              # => [3, 4, 5]
debug (foldl (acc x -> acc + x) 0 nums)       # => 15
debug (foldr (x acc -> x - acc) 0 [1, 2, 3])  # => 2

# pattern-lambdas: a lambda param may be a pattern (here, destructuring a pair)
debug (map ([a, b] -> a + b) [[1, 2], [3, 4]])  # => [3, 7]

# ---------------------------------------------------------
# 12. STRINGS, CHARS, FLOATS, BOOLS, CONVERSION
# ---------------------------------------------------------
debug "a string"
debug 'c'
debug 3.14
debug true
debug false
print (intToString 42)    # => 42       (strings print as raw text)
print (true ? "yes" : "no")  # => yes   (the ternary yields a string)
debug (1 == 1)            # => true
debug (1 < 2)             # => true
debug ("a" != "b")        # => true

# ---------------------------------------------------------
# 13. TYPE ANNOTATIONS — accepted and ignored at runtime
# ---------------------------------------------------------
debug {
    x: Int = 20
    x * 2
}                                   # => 40
debug {
    x: Int = 5
    x
}                                   # => 5

# ---------------------------------------------------------
# 14. ITERATION (for), pairs, format
#    for : fn -> seq -> unit   runs fn for its effect; result is discarded.
#    A list iterates per element; a record per entry as [field, value].
# ---------------------------------------------------------
for (x -> debug x) [1, 2, 3]                 # prints 1, 2, 3
[1, 2, 3] $ for (x -> debug (x * 10))        # prints 10, 20, 30

# records iterate as [field, value] pairs — destructure with a pattern-lambda;
# `format` builds a string, so `print` emits it as raw text
for ([f, v] -> print (format "%: %" [f, v])) [name: "Alice", age: 30]
                                             # prints name: Alice, age: 30

# like map, for is strictly fn-first:   for f seq  ==  seq $ for f
# (positioning matches map; the pipe feeds the collection in as the last arg)
[x: 1, y: 2, z: 3] $ for ([field, value] -> format "%: %" [field, value] $ print)
                                             # prints x: 1, y: 2, z: 3

# `for` is the one backtick that inverts (unlike `f`, which is f a b):
#   seq `for` fn  ==  seq $ for fn   (the sequence sits on the left)
[x: 1, y: 2, z: 3] `for` [field, value] -> format "%: %" [field, value] $ print
                                             # prints x: 1, y: 2, z: 3

pairs [a: 1, b: 2] $ debug                   # => [["a", 1], ["b", 2]]

print (format "% / %" [1, 2])                # => 1 / 2
print (format "last: %" [9])                 # => last: 9
print (format "100%%" [])                    # => 100%  (%% is a literal %)

# ---------------------------------------------------------
# 15. OUTPUT & CONVERSION — print / toString / parse / debug
#    print s      string only            => raw text, no quotes
#    toString x   any value -> string    => representation (strings stay quoted)
#    parse s      string -> value        => int | float | bool, else error
#    debug x      any value              => prints the representation (old print)
# ---------------------------------------------------------
print "a string"                       # => a string      (raw text)
debug "a string"                       # => "a string"    (debug shows quotes)
print (toString 42)                    # => 42
print (toString "hi")                  # => "hi"  (strings render quoted)
print (toString [1, 2, 3])             # => [1, 2, 3]
print (toString Nothing)               # => Nothing   (atoms render bare)
parse "42" + 8 $ intToString $ print   # => 50
debug (parse "3.5")                    # => 3.5
debug (parse "true")                   # => true

# ---------------------------------------------------------
# 16. GROUPS, BLOCKS, INDENTATION & SEMICOLONS
#    ( ) and { } are interchangeable: both are groups/blocks.
#    ()  or  { }  alone is unit. A block's value is the LAST statement's value.
#    Newlines act as implicit semicolons; an explicit `;` also separates
#    statements. Indentation is rigid: every nested level is exactly +4.
# ---------------------------------------------------------
debug ()                     # => ()            (empty group = unit)
debug {}                     # => ()            (empty block = unit)
debug {5}                    # => 5             (braces group like parens)
debug (5)                    # => 5             (parens group like braces)
debug (4 * 3)                # => 12
debug {4 * 3}                # => 12

group = { 1; 2; 3 }          # value of a block is its last statement
debug group                  # => 3

inlineBlock = { 1 + 1 }      # a same-line { } block
debug inlineBlock            # => 2

# statements on one line separated by explicit `;`
print "hello"; print ", world"        # prints: hello
                                      # prints: , world

# a multi-line `{ }` block as a function body (value = last statement)
area w h = {
    debug "computing area"
    w * h
}
debug (area 3 4)             # prints: computing area
                             # => 12

# local definitions are allowed inside a block
both = {
    x = 40
    y = 2
    x + y
}
debug both                   # => 42

# blank lines inside a block are fine; statements still stop at same-level
# newlines (implicit semicolons) and continue on +4-indented lines
seqOf = {
    debug "one"

    6 * 7
}
debug seqOf                  # prints: one
                             # => 42

# Whitespace is significant: a def cut off by a same-level newline binds UNIT,
# and the next line is a separate statement. So  func =\n{ ... }
# is  func = unit  PLUS a standalone block statement, NOT a function body.
unitAfter =
{
    debug "the brace block ran as its OWN statement"
}
debug (toString unitAfter)   # prints: the brace block ran as its OWN statement
                             # => ()      (unitAfter is unit, not a block)

# a function body that continues on a +4-indented line spans lines;
# `[ ]` also spans lines and elements separate by newline (commas optional),
# so arguments can be written across lines and even in columns:
spanArgs =
    myApp [
        1
        2
    ] [
        3
        4
    ]
debug (spanArgs)              # => [1, 2, 3, 4]

# INVALID (the lexer rejects these; demoed in comments only):
#   - a { that ends its own line with a body NOT indented by exactly +4:
#         func = {
#       print 1                #   <- must be indented exactly 4 spaces
#         }
#   - any indentation that skips a level (0 -> 8) or is not a multiple of 4.

# ---------------------------------------------------------
# 17. EXIT, LOAD & HALT
#    exit n    ends the whole process with status code n (n must be an int);
#              works anywhere: top level, inside blocks, lambdas, for, ...
#    load p    imports a file RELATIVE TO THE CURRENT FILE. Two effects:
#              (1) as a STATEMENT it merges the file's own definitions into
#                  the current scope (an `include`);
#              (2) as a VALUE it returns them as a MODULE record, so
#                  core = load "core.txt"; core.sq 4 ...  is namespaced access.
#              Inside a { } block the merge stays scoped to that block.
#    halt      is NOT a builtin. It is defined in core.txt — the standard
#              library — as  `halt = exit 0`. 0-arg value defs are LAZY, so
#              importing core.txt doesn't terminate the program; `halt` only
#              fires (exits 0) when it is actually USED.
# ---------------------------------------------------------
load "core.txt"             # include: sq and halt join THIS scope
debug (sq 5)                # => 25

core = load "core.txt"      # namespaced module record
debug (core.sq 4)           # => 16

# a load inside a { } block stays in the block's scope
scoped = {
    load "core.txt"
    sq 6
}
debug (scoped)              # => 36

debug "still running before the final halt"

# Laziness gotcha: a 0-arg def's { } body does NOT run at definition — it
# runs the first time the name is FORCED. So nothing prints above until here:
lazyNote = {
    debug "lazyNote WAS forced"
    6 * 7
}
debug (lazyNote * 2)      # forcing it prints "lazyNote WAS forced" (side effect)
                          # then the block's value:          # => 84
                          # A TRAILING top-level def is ALSO forced by file
                          # mode's final value pass (its block runs at the
                          # very end of the file); mid-file defs stay latent
                          # until used, exactly like lazyNote did above.
halt   # `halt = exit 0` — terminates the showcase cleanly (status 0)
